async function request(url, options = {}) {
    const response = await fetch(url, {
        ...options,
        headers: {
            "Authorization": "Bearer " + ACCESS_TOKEN,
            ...options.headers
        }
    });

    if (!response.ok) {
        const error = await response.text();
        throw new Error(error);
    }

    return response;
}

document.getElementById("uploadForm")
.addEventListener("submit", async function(e) {
    e.preventDefault();

    const name = document.getElementById("name").value.trim();
    const file = document.getElementById("file").files[0];
    const message = document.getElementById("message");

    try {
        message.textContent = "Uploading...";

        const path = "/" + name.replace(/[\\/]/g, "_")
                     + "_" + file.name.replace(/[\\/]/g, "_");

        await request(
            "https://content.dropboxapi.com/2/files/upload",
            {
                method: "POST",
                headers: {
                    "Dropbox-API-Arg": JSON.stringify({
                        path: path,
                        mode: "add",
                        autorename: true
                    }),
                    "Content-Type": "application/octet-stream"
                },
                body: file
            }
        );

        message.textContent = "Upload successful!";
        await loadFiles();

    } catch (error) {
        message.textContent = "Upload error: " + error.message;
    }
});

async function loadFiles() {
    const list = document.getElementById("fileList");
    const message = document.getElementById("message");

    list.innerHTML = "";
    message.textContent = "Loading...";

    try {
        const response = await request(
            "https://api.dropboxapi.com/2/files/list_folder",
            {
                method: "POST",
                headers: {
                    "Content-Type": "application/json"
                },
                body: JSON.stringify({
                    path: "",
                    recursive: false
                })
            }
        );

        const data = await response.json();
        const files = data.entries.filter(
            f => f[".tag"] === "file"
        );

        if (files.length === 0) {
            message.textContent = "No files found.";
            return;
        }

        files.forEach(function(f) {
            const li = document.createElement("li");
            li.appendChild(document.createTextNode(f.name + " "));

            const button = document.createElement("button");
            button.textContent = "Download";

            button.onclick = function() {
                downloadFile(f.path_lower);
            };

            li.appendChild(button);
            list.appendChild(li);
        });

        message.textContent = "Files loaded.";

    } catch (error) {
        message.textContent = "List error: " + error.message;
    }
}

async function downloadFile(path) {
    const message = document.getElementById("message");

    try {
        message.textContent = "Downloading...";

        const response = await request(
            "https://content.dropboxapi.com/2/files/download",
            {
                method: "POST",
                headers: {
                    "Dropbox-API-Arg": JSON.stringify({
                        path: path
                    })
                }
            }
        );

        const blob = await response.blob();
        const url = URL.createObjectURL(blob);
        const a = document.createElement("a");

        a.href = url;
        a.download = path.split("/").pop();
        document.body.appendChild(a);
        a.click();
        a.remove();

        setTimeout(() => URL.revokeObjectURL(url), 1000);

        message.textContent = "Download started.";

    } catch (error) {
        message.textContent = "Download error: " + error.message;
    }
}
